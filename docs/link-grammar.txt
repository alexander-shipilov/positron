link:
  | "{@link" /\s+/ link-ref "}"

link-ref:
  | entity
  | package-path entity

package-path:
  | "@" path-segment "/" path
  | path

path:
  | path-segment "/" path
  | path-segment "#"

path-segment:
  | /[a-z0-9._-]+/i

entity:
  | typed-ref "." entity
  | ref

typed-ref:
  | identifier
  | "(" identifier ":" ref-type ")"

ref:
  | typed-ref
  | "(" identifier ":" ref-index ")"

ref-type:
  | "class"
  | "enum"
  | "function"
  | "interface"
  | "namespace"
  | "type"
  | "variable"

ref-index:
  | /[1-9][0-9]?/

identifier:
  | /[_$\p{IDS}][$\p{IDC}]*/u
